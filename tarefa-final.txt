#!/usr/bin/env python3
"""Converte a Bíblia em USFM (eBible.org, porbr2018) em content-src/ do app.
Uso: python3 scripts/import-ebible.py <pasta-com-usfm> [--raiz .]
Falha (código 1) se faltar livro/capítulo; lacunas de versículo vão para docs/importacao.txt."""
import sys, re, json, shutil
from pathlib import Path

CODES = ("GEN EXO LEV NUM DEU JOS JDG RUT 1SA 2SA 1KI 2KI 1CH 2CH EZR NEH EST JOB PSA PRO ECC SNG ISA JER LAM EZK DAN HOS JOL AMO OBA JON MIC NAM HAB ZEP HAG ZEC MAL "
         "MAT MRK LUK JHN ACT ROM 1CO 2CO GAL EPH PHP COL 1TH 2TH 1TI 2TI TIT PHM HEB JAS 1PE 2PE 1JN 2JN 3JN JUD REV").split()

         28,16,24,21,28,16,16,13,6,6,4,4,5,3,6,4,3,1,13,5,5,3,5,1,1,1,22]

         "Mateus","Marcos","Lucas","João","Atos","Romanos","1 Coríntios","2 Coríntios","Gálatas","Efésios","Filipenses","Colossenses","1 Tessalonicenses","2 Tessalonicenses","1 Timóteo","2 Timóteo","Tito","Filemom","Hebreus","Tiago","1 Pedro","2 Pedro","1 João","2 João","3 João","Judas","Apocalipse"]
assert len(CODES) == 66 and len(CHAPS) == 66 and len(NAMES) == 66, (len(CODES), len(CHAPS), len(NAMES))

NOTE = re.compile(r'\\(f|fe|x|xt|ef|ex)\b.*?\\\1\*', re.S)
SKIP = {'id','ide','h','toc1','toc2','toc3','mt','mt1','mt2','mt3','ms','ms1','mr','s','s1','s2','s3','r','d','sp','rem','b','cl','cp','cd','sr','imt','is','ip','iot','io','ili','ie'}

def clean(t):
    t = NOTE.sub('', t)
    t = re.sub(r'\\w\s+([^|\\]*)\|[^\\]*\\w\*', r'\1', t)   # \w palavra|lemma\w*
    t = re.sub(r'\\\+?\w+\*?', '', t)          # demais marcadores de caractere (\add, \nd, \wj ...)
    return re.sub(r'\s+', ' ', t).strip()

def parse(path):
    raw = path.read_text(encoding='utf-8-sig')
    raw = NOTE.sub('', raw)
    m = re.search(r'^\\id\s+(\w{3})', raw, re.M)
    if not m: return None
    code, chapters, ch, vs = m.group(1).upper(), {}, None, None
    for line in raw.splitlines():
        mm = re.match(r'\\(\w+)\*?\s?(.*)$', line.strip())
        if not mm: 
            if vs is not None and line.strip(): vs[1].append(line.strip())
            continue
        tag, rest = mm.group(1), mm.group(2)
        if tag == 'c':
            n = re.match(r'(\d+)', rest)
            if n: ch = int(n.group(1)); chapters[ch] = []; vs = None
        elif tag == 'v' and ch is not None:
            n = re.match(r'(\d+)(?:[-–](\d+))?[a-b]?\s*(.*)$', rest)
            if n:
                vs = [int(n.group(1)), [n.group(3)], int(n.group(2)) if n.group(2) else None]
                chapters[ch].append(vs)
        elif tag in SKIP: 
            if tag in ('s','s1','s2','s3','r','d','b','ms','ms1','mr','sp'): vs = vs if tag == 'b' else vs
            continue
        elif vs is not None and rest:
            vs[1].append(rest)
    return code, chapters

def main():
    args = [a for a in sys.argv[1:] if not a.startswith('--')]
    src = Path(args[0]); root = Path(sys.argv[sys.argv.index('--raiz')+1]) if '--raiz' in sys.argv else Path('.')
    found = {}
    for f in sorted(list(src.rglob('*.usfm')) + list(src.rglob('*.SFM')) + list(src.rglob('*.sfm'))):
        r = parse(f)
        if r and r[0] in CODES: found[r[0]] = r[1]
    erros, avisos = [], []
    out = root / 'content-src' / 'biblia'
    if out.exists(): shutil.rmtree(out)
    total = 0
    for i, code in enumerate(CODES):
        bid = str(i + 1)
        if code not in found: erros.append(f'livro ausente: {NAMES[i]} ({code})'); continue
        chs = found[code]
        for c in range(1, CHAPS[i] + 1):
            verses = []
            for n, parts, ate in chs.get(c, []):
                t = clean(' '.join(parts))
                if t: verses.append({'n': n, 't': t})
            if not verses: erros.append(f'capítulo vazio: {NAMES[i]} {c}'); continue
            nums = [v['n'] for v in verses]
            if nums != sorted(set(nums)): erros.append(f'versículos fora de ordem: {NAMES[i]} {c}')
            falt = [k for k in range(1, max(nums)+1) if k not in nums]
            if falt: avisos.append(f'{NAMES[i]} {c}: faltam versículos {falt}')
            d = out / bid; d.mkdir(parents=True, exist_ok=True)
            (d / f'{bid}-{c}.json').write_text(json.dumps({'book': bid, 'chapter': c, 'verses': verses}, ensure_ascii=False, separators=(',', ':')), encoding='utf-8')
            total += 1
        extra = [c for c in chs if c > CHAPS[i]]
        if extra: erros.append(f'capítulos a mais em {NAMES[i]}: {extra}')
    docs = root / 'docs'; docs.mkdir(exist_ok=True)
    (docs / 'importacao.txt').write_text('Capítulos gravados: %d de 1189\nERROS (%d):\n%s\nAVISOS (%d):\n%s\n' % (total, len(erros), '\n'.join(erros), len(avisos), '\n'.join(avisos)), encoding='utf-8')
    if '--producao' in sys.argv and not erros:
        for t in (root/'content-src'/'camada2'/'teste', ):
            if t.exists(): shutil.rmtree(t)
        mp = root/'content-src'/'meta.json'; meta = json.loads(mp.read_text(encoding='utf-8'))
        meta.update(contentVersion='1.0.0', production=True,
                    license={'text': 'Texto bíblico: Bíblia Livre (BLIVRE), 2018, de Diego Santos, Mario Sérgio e Marco Teles, licença Creative Commons Atribuição 4.0 (CC BY 4.0). Fonte: eBible.org. O app não altera o texto.', 'url': 'https://ebible.org/porbr2018/copyright.htm'},
                    books=[{'id': str(i+1), 'name': NAMES[i]} for i in range(66)])
        mp.write_text(json.dumps(meta, ensure_ascii=False, separators=(',', ':')) + '\n', encoding='utf-8')
    print('capítulos', total, 'erros', len(erros), 'avisos', len(avisos))
    if erros: print('\n'.join(erros[:20])); sys.exit(1)

main()

# Palavra Viva

Bíblia em português, gratuita e sem internet, com ensino em camadas. PWA estático publicado no GitHub Pages.

## Como o app funciona
- **Estrutura fixa:** `src/books.js` tem os 66 livros. O texto vem de pacotes em `content/`, listados em `content-manifest.json` com hash SHA-256. Pacote que não confere com o hash é recusado.
- **Offline:** o service worker (`sw.js`) guarda o app e os capítulos na instalação. "Baixar para ler sem internet" guarda também as passagens de estudo.
- **Atualização:** nunca automática. O app mostra "Há uma versão nova" e só troca quando a pessoa toca em Atualizar. O número de versão do app é carimbado no deploy (hash dos arquivos do app), então só muda quando o app muda.
- **Conteúdo:** só aceita manifesto de geração maior que a guardada. O último conteúdo válido nunca é descartado.
- **Estudo:** abas Ler, Entender, Ligar ao evangelho e Aprofundar. Aba sem conteúdo não aparece.

## Configuração (`config.json`)
`ownerName`, `year`, `contactEmail`, `pixKey`, `siteUrl`, `audio`. Se `pixKey` estiver vazio, o item "Abençoe este Projeto" some. A chave Pix nunca aparece na tela, só é copiada.

## Arquivos que ficam fora do código (enviar para `assets/`)
`hero-bible.jpg`, `hero-dawn.jpg`, `hero-reading.jpg`, `abencoe.jpg`, `ambiente-leitura.mp3` (opcional: sem ele o app usa um pad sintetizado). Se faltar imagem, aparece um fundo verde no lugar.

## Testes
- `node --test tests/*.test.mjs` e `node scripts/validate-content.mjs` (rodam no CI a cada push).

## Adicionar conteúdo
1. Coloque o JSON em `content-src/biblia/...` (capítulo) ou `content-src/camada2/...` (passagem).
2. `node scripts/build-manifest.mjs` (gera arquivos com hash e sobe a geração).
3. `node scripts/validate-content.mjs`, depois commit e push.

Passagens reais exigem auditoria: 2 auditores de famílias diferentes entre si e do gerador, 36 critérios todos "P". Enquanto `production` for `false` em `content-src/meta.json`, o app avisa que o conteúdo é fictício.

## Numeração do conteúdo
`minGeneration` em `content-src/meta.json` garante que a geração nunca fique abaixo da que os aparelhos já guardaram (o app só aceita geração maior). Não diminua esse número.

## Voltar atrás (rollback)
Restaure em `content-src/` a versão boa e rode o build: a geração sobe de novo (nunca desce), então os aparelhos aceitam.

## Fábrica de conteúdo (Kaggle)
`.github/workflows/kaggle-viabilidade.yml` testa se o Kaggle libera GPU e internet pela API e quais modelos baixam sem aceitar licença. O resultado fica em `docs/viabilidade/`. Precisa do segredo `KAGGLE_API_TOKEN`.

  async function downloadAll(cb) {
    var c = await caches.open(CONTENT);
    var have = (await c.keys()).map(function (r) { return relPath(r.url); });
    var need = core.missingPackages(manifest, have), total = manifest.packages.length, done = total - need.length, fails = 0, i = 0;
    cb(done, total);
    async function worker() {
      while (i < need.length) {
        var e = need[i++];
        try {
          var r = await fetch(e.path, { cache: 'no-store' }); if (!r.ok) throw new Error('http');
          var bytes = new Uint8Array(await r.clone().arrayBuffer());
          if (!(await core.verifyPackage(bytes, e)).ok) throw new Error('hash');
          await c.put(e.path, r); done++;
        } catch (x) { fails++; }
        cb(done, total);
      }
    }
    await Promise.all([worker(), worker(), worker(), worker()]);
    for (var m = 0; m < (cfg.audio || []).length; m++) { try { await mediaResponse(cfg.audio[m]); } catch (x) { /* áudio é opcional */ } }
    if (!fails) { kvSet('baixado', { generation: manifest.generation, at: Date.now() }); try { if (navigator.storage && navigator.storage.persist) navigator.storage.persist(); } catch (x) { /* ignora */ } }
    return { done: done, total: total, fails: fails };
  }

  function startSynth() {
    var C = window.AudioContext || window.webkitAudioContext; if (!C) return false;
    aud.ctx = aud.ctx || new C(); if (aud.ctx.state === 'suspended') aud.ctx.resume();
    var ac = aud.ctx; aud.master = ac.createGain(); aud.master.gain.value = 0;
    var lp = ac.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 900; aud.master.connect(lp); lp.connect(ac.destination);
    [130.81, 196, 261.63, 329.63].forEach(function (f, i) {
      var o = ac.createOscillator(); o.type = 'sine'; o.frequency.value = f;
      var g = ac.createGain(); g.gain.value = 0.12 / (1 + i * 0.4);
      var l = ac.createOscillator(); l.frequency.value = 0.07 + i * 0.03; var lg = ac.createGain(); lg.gain.value = 0.05; l.connect(lg); lg.connect(g.gain);
      o.connect(g); g.connect(aud.master); o.start(); l.start(); aud.oscs.push(o, l);
    });
    aud.master.gain.linearRampToValueAtTime(volume() * 0.8, ac.currentTime + 2.5);
    return true;
  }
  async function startAudio() {
    var src = cfg.audio && cfg.audio[0];

    }
  }

/* Mini-validador de JSON Schema (subconjunto usado em schemas/). Sem dependências. */
export function check(schema, value, path = '$') {
  const errs = [];
  const t = schema.type;
  const typeOk = { object: v => v && typeof v === 'object' && !Array.isArray(v), array: Array.isArray, string: v => typeof v === 'string', integer: Number.isInteger, boolean: v => typeof v === 'boolean' };
  if (t && !typeOk[t](value)) return [`${path}: tipo esperado ${t}`];
  if (t === 'string') {
    if (schema.minLength && value.length < schema.minLength) errs.push(`${path}: texto curto demais`);
    if (schema.pattern && !new RegExp(schema.pattern).test(value)) errs.push(`${path}: formato inválido`);
  }
  if (t === 'integer' && schema.minimum !== undefined && value < schema.minimum) errs.push(`${path}: menor que ${schema.minimum}`);
  if (t === 'array') {
    if (schema.minItems && value.length < schema.minItems) errs.push(`${path}: itens insuficientes`);
    if (schema.items) value.forEach((x, i) => errs.push(...check(schema.items, x, `${path}[${i}]`)));
  }
  if (t === 'object') {
    for (const k of schema.required || []) if (!(k in value)) errs.push(`${path}.${k}: obrigatório`);
    for (const [k, s] of Object.entries(schema.properties || {})) if (k in value) errs.push(...check(s, value[k], `${path}.${k}`));
  }
  return errs;
}

/* Valida content-manifest.json e content/ . Sai com código 1 se algo estiver errado. */
import { readFileSync, existsSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { join, dirname, basename } from 'node:path';
import { fileURLToPath } from 'node:url';
import { check } from './mini-schema.mjs';
const root = join(dirname(fileURLToPath(import.meta.url)), '..');
const J = p => JSON.parse(readFileSync(join(root, p), 'utf8'));
export function validate(rootDir = root) {
  const R = p => JSON.parse(readFileSync(join(rootDir, p), 'utf8'));
  const errs = [];
  const m = R('content-manifest.json');
  errs.push(...check(J('schemas/manifest.schema.json'), m, 'manifesto'));
  if (errs.length) return errs;
  const seen = new Set();
  for (const p of m.packages) {
    if (seen.has(p.path)) errs.push('caminho duplicado: ' + p.path); seen.add(p.path);
    const f = join(rootDir, p.path);
    if (!existsSync(f)) { errs.push('arquivo ausente: ' + p.path); continue; }
    const bytes = readFileSync(f);
    const h = createHash('sha256').update(bytes).digest('hex');
    if (h !== p.sha256) errs.push('hash diferente do manifesto: ' + p.path);
    if (!basename(p.path).includes('.' + h.slice(0, 12) + '.')) errs.push('hash no nome não confere: ' + p.path);
    if (bytes.length !== p.size) errs.push('tamanho diferente: ' + p.path);
    let d; try { d = JSON.parse(bytes.toString('utf8')); } catch { errs.push('JSON inválido: ' + p.path); continue; }
    const isTest = d._teste === true;
    if (isTest !== (p.teste === true)) errs.push('marca de teste incoerente: ' + p.path);
    if (isTest && m.production) errs.push('produção com conteúdo de teste: ' + p.path);
    if (p.kind === 'biblia') errs.push(...check(J('schemas/chapter.schema.json'), d, p.path));
    else {
      errs.push(...check(J('schemas/passage.schema.json'), d, p.path));
      if (!isTest) {
        const a = d.audit;
        if (!a) errs.push('passagem real sem auditoria: ' + p.path);
        else {
          const fams = a.auditors.map(x => x.family);
          if (new Set(fams).size < 2 || fams.includes(a.generator.family)) errs.push('auditores sem famílias diferentes entre si e do gerador: ' + p.path);
          const gates = Object.values(a.gates || {});
          if (gates.length !== 36 || gates.some(g => g !== 'P')) errs.push('os 36 critérios precisam estar todos P: ' + p.path);
        }
      }
    }
  }
  for (const b of m.bible) for (const c of b.chapters) if (!seen.has(c.path)) errs.push('capítulo sem pacote: ' + c.path);
  const ids = new Set();
  for (const ps of m.passages || []) { if (ids.has(ps.id)) errs.push('id repetido: ' + ps.id); ids.add(ps.id); if (!seen.has(ps.path)) errs.push('passagem sem pacote: ' + ps.id); }
  return errs;
}
if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const e = validate();
  if (e.length) { console.error(e.join('\n')); process.exit(1); }
  console.log('conteúdo válido');
}

/* Núcleo do Palavra Viva: lógica pura, usada pelo app, pelo service worker e pelos testes.
   Sem rede, sem dependências. Regras da seção 5 do plano. */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.PVCore = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';
  const PREFIX = 'pv4-';

  async function sha256Hex(bytes) {
    const buf = await crypto.subtle.digest('SHA-256', bytes);
    return Array.from(new Uint8Array(buf)).map(function (b) { return b.toString(16).padStart(2, '0'); }).join('');
  }

  /* Confere um pacote baixado contra a entrada do manifesto. Falha fechada. */
  async function verifyPackage(bytes, entry) {
    if (!entry || typeof entry.sha256 !== 'string') return { ok: false, reason: 'entrada_invalida' };
    const h = await sha256Hex(bytes);
    if (h !== entry.sha256) return { ok: false, reason: 'hash_diferente' };
    if (typeof entry.size === 'number' && bytes.byteLength !== entry.size) return { ok: false, reason: 'tamanho_diferente' };
    return { ok: true };
  }

  /* Decide se um manifesto novo é aceito e o que remover do cache.
     Aceita só geração MAIOR (rollback também incrementa a geração).
     Remove do cache o que não consta no manifesto novo e o que foi revogado. */
  function planUpdate(current, incoming, cachedPaths) {
    if (!incoming || !Number.isInteger(incoming.generation) || !Array.isArray(incoming.packages)) {
      return { accept: false, reason: 'manifesto_invalido', remove: [] };
    }
    if (current && Number.isInteger(current.generation)) {
      if (incoming.generation < current.generation) return { accept: false, reason: 'geracao_menor', remove: [] };
      if (incoming.generation === current.generation) return { accept: false, reason: 'mesma_geracao', remove: [] };
    }
    const keep = new Set(incoming.packages.map(function (p) { return p.path; }));
    const revoked = new Set(incoming.revoked || []);
    const shaByPath = new Map();
    ((current && current.packages) || []).forEach(function (p) { shaByPath.set(p.path, p.sha256); });
    incoming.packages.forEach(function (p) { shaByPath.set(p.path, p.sha256); });
    const remove = (cachedPaths || []).filter(function (p) {
      return !keep.has(p) || revoked.has(shaByPath.get(p));
    });
    return { accept: true, reason: 'ok', remove: remove };
  }

  /* Nomes de cache a apagar: só os do app (prefixo pv4-) que não estão na lista de mantidos. */
  function cleanupPlan(cacheNames, keep) {
    const k = new Set(keep || []);
    return (cacheNames || []).filter(function (n) { return n.indexOf(PREFIX) === 0 && !k.has(n); });
  }

  /* Abas visíveis de uma passagem: aba sem conteúdo nunca aparece; nunca "em breve". */
  function visibleTabs(passage) {
    const tabs = [];
    if (!passage) return tabs;
    const has = function (v) { return typeof v === 'string' && v.trim().length > 0; };
    if (Array.isArray(passage.verses) && passage.verses.length) tabs.push('ler');
    if (has(passage.entender)) tabs.push('entender');
    if (passage.ligar && has(passage.ligar.texto)) tabs.push('ligar');
    if (Array.isArray(passage.refs) && passage.refs.length) tabs.push('aprofundar');
    return tabs;
  }


  /* Texto sem acento e em minúsculas, para busca. */
  function normalize(s) {
    return String(s == null ? '' : s).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
  }

  /* Acha a consulta no texto ignorando acento e caixa. Devolve {start,end} nos índices do texto ORIGINAL, ou null. */
  function findMatch(text, query) {
    const q = normalize(query).trim();
    if (!q) return null;
    let nt = '';
    const map = [];
    const src = String(text == null ? '' : text);
    for (let i = 0; i < src.length; i++) {
      const n = normalize(src[i]);
      for (let k = 0; k < n.length; k++) { nt += n[k]; map.push(i); }
    }
    const idx = nt.indexOf(q);
    if (idx < 0) return null;
    return { start: map[idx], end: map[idx + q.length - 1] + 1 };
  }

  /* Separa "Livro de Teste 1:2" ou "João 3:16" em {name, chapter, verse}. */
  function parseRef(ref) {
    const m = /^\s*(.+?)\s+(\d+)(?::(\d+))?(?:\s*[-–]\s*\d+)?\s*$/.exec(String(ref || ''));
    if (!m) return null;
    return { name: m[1], chapter: parseInt(m[2], 10), verse: m[3] ? parseInt(m[3], 10) : null };
  }

  /* Selo de autoria do ensino. Texto fixo exigido pelo plano; "Revisado por pessoa" só se humanReviewed === true. */
  function stampText(passage) {
    if (!passage) return '';
    if (passage._teste === true) return 'Exemplo de teste: este texto é fictício.';
    let t = 'Ensino do app, não é a Bíblia. Gerado por IA e conferido por outras IAs.';
    if (passage.audit && passage.audit.humanReviewed === true) t += ' Revisado por pessoa.';
    return t;
  }

  /* Cronograma de download: lista de pacotes que faltam no cache. */
  function missingPackages(manifest, cachedPaths) {
    const have = new Set(cachedPaths || []);
    return ((manifest && manifest.packages) || []).filter(function (p) { return !have.has(p.path); });
  }

  return { normalize: normalize, findMatch: findMatch, parseRef: parseRef, stampText: stampText, missingPackages: missingPackages, PREFIX: PREFIX, sha256Hex: sha256Hex, verifyPackage: verifyPackage, planUpdate: planUpdate, cleanupPlan: cleanupPlan, visibleTabs: visibleTabs };
});

/* Service worker do Palavra Viva.
   - sem skipWaiting automático: a página pede com o botão "Atualizar";
   - cache antigo só é apagado na ativação do novo, e só chaves com prefixo pv4-;
   - sw.js e content-manifest.json sempre da rede (no-store); cache só como reserva offline;
   - __SHELL_HASH__ é trocado no deploy pelo hash dos arquivos do app: só muda quando o app muda. */
importScripts('src/core.js');
const VERSION = '__SHELL_HASH__';
const SHELL = 'pv4-shell-' + VERSION;
const CONTENT = 'pv4-content';
const MEDIA = 'pv4-media';
const SHELL_FILES = ['./', 'index.html', 'app.js', 'styles.css', 'src/core.js', 'src/books.js', 'manifest.json', 'reset.html', 'icons/icon-192.png', 'icons/icon-512.png'];
/* opcionais: se faltarem no servidor, a instalação NÃO falha */
const OPTIONAL_FILES = ['config.json', 'assets/hero-bible.jpg', 'assets/hero-dawn.jpg', 'assets/hero-reading.jpg', 'assets/abencoe.jpg'];

async function precacheContent() {
  const r = await fetch('content-manifest.json', { cache: 'no-store' });
  if (!r.ok) throw new Error('manifesto indisponível');
  const text = await r.text();
  const m = JSON.parse(text);
  const c = await caches.open(CONTENT);
  for (const e of m.packages.filter(function (p) { return p.precache; })) {
    if (await c.match(e.path)) continue; // já guardado e verificado
    const pr = await fetch(e.path, { cache: 'no-store' });
    if (!pr.ok) throw new Error('pacote indisponível: ' + e.path);
    const bytes = new Uint8Array(await pr.clone().arrayBuffer());
    const v = await PVCore.verifyPackage(bytes, e);
    if (!v.ok) throw new Error('hash não confere: ' + e.path); // falha fechada: instalação não conclui
    await c.put(e.path, pr);
  }
  await c.put('content-manifest.json', new Response(text, { headers: { 'content-type': 'application/json' } }));
}

self.addEventListener('install', function (event) {
  event.waitUntil((async function () {
    const c = await caches.open(SHELL);
    await c.addAll(SHELL_FILES.map(function (u) { return new Request(u, { cache: 'reload' }); }));
    await Promise.all(OPTIONAL_FILES.map(async function (u) {
      try { const r = await fetch(new Request(u, { cache: 'reload' })); if (r.ok) await c.put(u, r); } catch (e) { /* opcional */ }
    }));
    await precacheContent();
  })());
});

self.addEventListener('activate', function (event) {
  event.waitUntil((async function () {
    const names = await caches.keys();
    await Promise.all(PVCore.cleanupPlan(names, [SHELL, CONTENT, MEDIA]).map(function (n) { return caches.delete(n); }));
    await self.clients.claim();
  })());
});

self.addEventListener('message', function (event) {
  if (event.data && event.data.type === 'SKIP_WAITING') self.skipWaiting();
});

async function networkFirst(req, cacheName, key) {
  try {
    const r = await fetch(req, { cache: 'no-store' });
    if (r.ok) { const c = await caches.open(cacheName); await c.put(key, r.clone()); }
    return r;
  } catch (e) {
    const c = await caches.open(cacheName);
    const hit = await c.match(key);
    if (hit) return hit;
    throw e;
  }
}

self.addEventListener('fetch', function (event) {
  const req = event.request;
  if (req.method !== 'GET') return;
  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return;
  const file = url.pathname.split('/').pop();
  if (file === 'sw.js') return; // sempre da rede, nunca interceptado
  if (file === 'content-manifest.json') { event.respondWith(networkFirst(req, CONTENT, 'content-manifest.json')); return; }
  if (file === 'config.json') { event.respondWith(networkFirst(req, SHELL, 'config.json')); return; }
  event.respondWith((async function () {
    const hit = await caches.match(req, { ignoreSearch: true });
    if (hit) return hit;
    try {
      const r = await fetch(req);
      if (r.ok && r.status === 200 && /\/assets\//.test(url.pathname) && !req.headers.has('range')) { const c = await caches.open(SHELL); await c.put(req, r.clone()); }
      return r;
    } catch (e) {
      if (req.mode === 'navigate') { const shell = await caches.open(SHELL); const idx = await shell.match('index.html'); if (idx) return idx; }
      throw e;
    }
  })());
});
